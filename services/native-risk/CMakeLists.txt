cmake_minimum_required(VERSION 3.20)
project(n9n8_native_risk LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(n9n8_orderbook src/orderbook.cpp)
target_include_directories(n9n8_orderbook PUBLIC include)
target_compile_options(n9n8_orderbook PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(n9n8_orderbook_test tests/orderbook_test.cpp)
target_link_libraries(n9n8_orderbook_test PRIVATE n9n8_orderbook)
enable_testing()
add_test(NAME n9n8_orderbook_test COMMAND n9n8_orderbook_test)
